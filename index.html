        function toggleTheme() {

            if (document.documentElement.classList.contains('dark')) {

                document.documentElement.classList.remove('dark');

                localStorage.theme = 'light';

            } else {

                document.documentElement.classList.add('dark');

                localStorage.theme = 'dark';

            }

        }


        function getIcon(type) {

            switch(type) {

                case 'food': return '🍜';

                case 'hotel': return '🏨';

                case 'sightseeing': return '📸';

                case 'cruise': return '🚢';

                case 'booking': return '🎟️';

                case 'special': return '🎉';

                default: return '📍';

            }

        }


        function getBg(type) {

                case 'special': return 'bg-pink-50 dark:bg-pink-900/30 border-pink-200 dark:border-pink-800/50';

                case 'flight': return 'bg-blue-50 dark:bg-blue-900/30 border-blue-200 dark:border-blue-800/50';

                default: return 'bg-white dark:bg-slate-800 border-slate-200 dark:border-slate-700';

            }

        }

        function scrollToDate(dateStr) {

            const id = dateStr.replace(' ', '-');

            const el = document.getElementById(id);

            if (el) {

                const yOffset = -100; // Account for the sticky header

                const y = el.getBoundingClientRect().top + window.pageYOffset + yOffset;

                window.scrollTo({top: y, behavior: 'smooth'});

            }

        }


        function scrollToToday() {

            const today = new Date();

            const monthNames = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];

            

            // Format current date to match "April 14" format

            const currentMonthStr = monthNames[today.getMonth()];

            const currentDateStr = `${currentMonthStr} ${today.getDate()}`;

            

            // Check if today is a date listed in the itinerary

            const targetId = currentDateStr.replace(' ', '-');

            

            if (document.getElementById(targetId)) {

                // Add a small delay so the user briefly sees the top before it jumps down automatically

                setTimeout(() => {

                    scrollToDate(currentDateStr);

                }, 800);

            }

            // If the date isn't found (e.g., viewing before the trip starts), it will naturally just stay at the top.

        }
